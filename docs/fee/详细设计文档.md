```java
package com.ruoyi.system.service.impl;

import cn.hutool.core.util.StrUtil;
import com.baomidou.mybatisplus.core.conditions.Wrapper;
import com.baomidou.mybatisplus.core.toolkit.Wrappers;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.ruoyi.common.core.domain.PageQuery;
import com.ruoyi.common.helper.LoginHelper;
import com.ruoyi.system.domain.FeeOperationLog;
import com.ruoyi.system.mapper.FeeOperationLogMapper;
import com.ruoyi.system.service.IFeeOperationLogService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;
import java.util.Date;
import java.util.List;

@Slf4j
@Service
@RequiredArgsConstructor
public class FeeOperationLogServiceImpl implements IFeeOperationLogService {

    private final FeeOperationLogMapper feeOperationLogMapper;

    @Override
    public void log(String module, String operationType, Long billId, String billNo,
                    Long relatedAppId, String beforeStatus, String afterStatus,
                    String operationDetail) {
        FeeOperationLog logEntry = new FeeOperationLog();
        logEntry.setModule(module);
        logEntry.setOperationType(operationType);
        logEntry.setBillId(billId);
        logEntry.setBillNo(billNo);
        logEntry.setRelatedAppId(relatedAppId);
        logEntry.setOperatorId(LoginHelper.getUserId());
        logEntry.setOperatorName(LoginHelper.getNickName());
        logEntry.setOperatorRole(LoginHelper.getLoginUser().getRoleKeys().toString());
        logEntry.setBeforeStatus(beforeStatus);
        logEntry.setAfterStatus(afterStatus);
        logEntry.setOperationDetail(operationDetail);
        logEntry.setOperationTime(new Date());
        feeOperationLogMapper.insert(logEntry);
    }

    @Override
    public Page<FeeOperationLog> selectPageList(FeeOperationLog query, PageQuery pageQuery) {
        Wrapper<FeeOperationLog> wrapper = buildQueryWrapper(query);
        return feeOperationLogMapper.selectPage(pageQuery.build(), wrapper);
    }

    @Override
    public List<FeeOperationLog> selectTimeline(String billNo) {
        return feeOperationLogMapper.selectList(
            Wrappers.<FeeOperationLog>lambdaQuery()
                .eq(FeeOperationLog::getBillNo, billNo)
                .orderByAsc(FeeOperationLog::getOperationTime));
    }

    @Override
    public List<FeeOperationLog> selectExportList(FeeOperationLog query) {
        return feeOperationLogMapper.selectList(buildQueryWrapper(query));
    }

    private Wrapper<FeeOperationLog> buildQueryWrapper(FeeOperationLog query) {
        return Wrappers.<FeeOperationLog>lambdaQuery()
            .eq(StrUtil.isNotBlank(query.getModule()), FeeOperationLog::getModule, query.getModule())
            .eq(StrUtil.isNotBlank(query.getOperationType()), FeeOperationLog::getOperationType, query.getOperationType())
            .eq(StrUtil.isNotBlank(query.getBillNo()), FeeOperationLog::getBillNo, query.getBillNo())
            .eq(query.getOperatorId() != null, FeeOperationLog::getOperatorId, query.getOperatorId())
            .orderByDesc(FeeOperationLog::getOperationTime);
    }
}
```

**文件**：`ruoyi-system/src/main/java/com/ruoyi/system/service/impl/ExpenseApprovalServiceImpl.java`

```java
package com.ruoyi.system.service.impl;

import cn.hutool.core.util.StrUtil;
import cn.hutool.json.JSONUtil;
import com.ruoyi.common.exception.ServiceException;
import com.ruoyi.common.helper.LoginHelper;
import com.ruoyi.system.domain.ExpenseBill;
import com.ruoyi.system.mapper.ExpenseBillMapper;
import com.ruoyi.system.service.*;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.util.Date;
import java.util.HashMap;
import java.util.Map;

@Slf4j
@Service
@RequiredArgsConstructor
public class ExpenseApprovalServiceImpl implements IExpenseApprovalService {

    private final ExpenseBillMapper expenseBillMapper;
    private final IFeeOperationLogService feeOperationLogService;
    private final ISysMessageService sysMessageService;

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Boolean approve(Long billId, String opinion) {
        ExpenseBill bill = expenseBillMapper.selectById(billId);
        if (bill == null) throw new ServiceException("报销单不存在");
        if (!"SUBMITTED".equals(bill.getStatus())) {
            throw new ServiceException("报销单当前状态不允许审批");
        }
        // 校验审批权限：仅部门经理或admin
        validateApprover(bill);

        String beforeStatus = bill.getStatus();
        bill.setStatus("APPROVED");
        bill.setApproverId(LoginHelper.getUserId());
        bill.setApproveTime(new Date());
        expenseBillMapper.updateById(bill);

        // 写操作日志
        Map<String, Object> detail = new HashMap<>();
        detail.put("opinion", opinion);
        detail.put("amount", bill.getAmount());
        feeOperationLogService.log("EXPENSE", "APPROVE", billId, bill.getBillNo(),
            null, beforeStatus, "APPROVED", JSONUtil.toJsonStr(detail));

        // 发送站内通知给报销人
        sysMessageService.sendMessage(bill.getApplicantId(), "报销单审批通过",
            "您的报销单（" + bill.getBillNo() + "）已通过审批，金额：¥" + bill.getAmount());

        return true;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Boolean reject(Long billId, String reason, String opinion) {
        if (StrUtil.isBlank(reason)) throw new ServiceException("驳回原因不能为空");
        if (reason.length() < 5 || reason.length() > 500) {
            throw new ServiceException("驳回原因需5-500字符");
        }

        ExpenseBill bill = expenseBillMapper.selectById(billId);
        if (bill == null) throw new ServiceException("报销单不存在");
        if (!"SUBMITTED".equals(bill.getStatus())) {
            throw new ServiceException("报销单当前状态不允许审批");
        }
        validateApprover(bill);

        String beforeStatus = bill.getStatus();
        bill.setStatus("REJECTED");
        bill.setRejectReason(reason);
        bill.setRejectTime(new Date());
        expenseBillMapper.updateById(bill);

        // 写操作日志
        Map<String, Object> detail = new HashMap<>();
        detail.put("rejectReason", reason);
        detail.put("opinion", opinion);
        feeOperationLogService.log("EXPENSE", "REJECT", billId, bill.getBillNo(),
            null, beforeStatus, "REJECTED", JSONUtil.toJsonStr(detail));

        // 通知报销人
        sysMessageService.sendMessage(bill.getApplicantId(), "报销单被驳回",
            "您的报销单（" + bill.getBillNo() + "）被驳回，原因：" + reason + "，请修改后重新提交。");

        return true;
    }

    private void validateApprover(ExpenseBill bill) {
        String roleKey = LoginHelper.getLoginUser().getRoleKeys().iterator().next();
        if ("admin".equals(roleKey)) return;
        if (!"dept_leader".equals(roleKey)) {
            throw new ServiceException("仅部门经理可执行审批操作");
        }
        // 校验审批人部门与报销人部门一致
        Long approverDeptId = LoginHelper.getDeptId();
        if (!approverDeptId.equals(bill.getDeptId())) {
            throw new ServiceException("您不是该报销人所属部门的部门经理");
        }
    }
}
```

```java
package com.ruoyi.system.service.impl;

import cn.hutool.core.util.StrUtil;
import cn.hutool.json.JSONUtil;
import com.baomidou.mybatisplus.core.conditions.Wrapper;
import com.baomidou.mybatisplus.core.toolkit.Wrappers;
import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
import com.ruoyi.common.core.domain.PageQuery;
import com.ruoyi.common.exception.ServiceException;
import com.ruoyi.common.helper.LoginHelper;
import com.ruoyi.system.domain.*;
import com.ruoyi.system.domain.vo.OverBudgetApprovalVo;
import com.ruoyi.system.mapper.*;
import com.ruoyi.system.service.*;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.math.BigDecimal;
import java.util.*;

@Slf4j
@Service
@RequiredArgsConstructor
public class OverBudgetApprovalServiceImpl implements IOverBudgetApprovalService {

    private final OverBudgetApprovalMapper overBudgetApprovalMapper;
    private final BudgetAdjustmentLogMapper adjustmentLogMapper;
    private final BudgetPreparationDetailMapper budgetDetailMapper;
    private final ExpenseBillMapper expenseBillMapper;
    private final IFeeOperationLogService feeOperationLogService;
    private final ISysMessageService sysMessageService;

    @Override
    public OverBudgetApprovalVo queryById(Long id) {
        return overBudgetApprovalMapper.selectVoById(id);
    }

    @Override
    public Page<OverBudgetApprovalVo> selectPageList(Long billId, String status, PageQuery pageQuery) {
        Wrapper<OverBudgetApproval> wrapper = Wrappers.<OverBudgetApproval>lambdaQuery()
            .eq(billId != null, OverBudgetApproval::getBillId, billId)
            .eq(StrUtil.isNotBlank(status), OverBudgetApproval::getStatus, status)
            .orderByDesc(OverBudgetApproval::getCreateTime);
        return overBudgetApprovalMapper.selectVoPage(pageQuery.build(), wrapper);
    }

    @Override
    public Page<OverBudgetApprovalVo> pendingList(PageQuery pageQuery) {
        Wrapper<OverBudgetApproval> wrapper = Wrappers.<OverBudgetApproval>lambdaQuery()
            .eq(OverBudgetApproval::getStatus, "PENDING")
            .orderByAsc(OverBudgetApproval::getCreateTime);
        return overBudgetApprovalMapper.selectVoPage(pageQuery.build(), wrapper);
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Boolean approve(Long id, String opinion) {
        OverBudgetApproval approval = overBudgetApprovalMapper.selectById(id);
        validatePending(approval);

        // 追加科目预算额度
        budgetDetailMapper.addBudgetBySubject(approval.getBudgetSubject(),
            approval.getCostCenterCode(), approval.getShortageAmount());

        approval.setApprovalAction("APPROVE");
        approval.setApprovalOpinion(opinion);
        approval.setApproverId(LoginHelper.getUserId());
        approval.setApproverName(LoginHelper.getNickName());
        approval.setApprovalTime(new Date());
        approval.setStatus("APPROVED");
        overBudgetApprovalMapper.updateById(approval);

        // 更新报销单状态
        ExpenseBill bill = expenseBillMapper.selectById(approval.getBillId());
        if (bill != null) {
            bill.setStatus("APPROVED");
            bill.setIsOverBudget(true);
            bill.setOverBudgetApprovalId(id);
            expenseBillMapper.updateById(bill);
        }

        // 操作日志
        Map<String, Object> detail = new HashMap<>();
        detail.put("shortageAmount", approval.getShortageAmount());
        detail.put("budgetSubject", approval.getBudgetSubject());
        feeOperationLogService.log("OVER_BUDGET", "OB_APPROVE", approval.getBillId(),
            approval.getBillNo(), id, "PENDING", "APPROVED", JSONUtil.toJsonStr(detail));

        // 通知报销人
        if (bill != null) {
            sysMessageService.sendMessage(bill.getApplicantId(), "超预算审批通过",
                "您的超预算报销单（" + approval.getBillNo() + "）已通过，追加预算 ¥" + approval.getShortageAmount());
        }
        return true;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Boolean reject(Long id, String reason) {
        if (StrUtil.isBlank(reason)) throw new ServiceException("驳回原因不能为空");
        OverBudgetApproval approval = overBudgetApprovalMapper.selectById(id);
        validatePending(approval);

        approval.setApprovalAction("REJECT");
        approval.setApprovalOpinion(reason);
        approval.setApproverId(LoginHelper.getUserId());
        approval.setApproverName(LoginHelper.getNickName());
        approval.setApprovalTime(new Date());
        approval.setStatus("REJECTED");
        overBudgetApprovalMapper.updateById(approval);

        // 操作日志
        feeOperationLogService.log("OVER_BUDGET", "OB_REJECT", approval.getBillId(),
            approval.getBillNo(), id, "PENDING", "REJECTED",
            JSONUtil.toJsonStr(Collections.singletonMap("rejectReason", reason)));

        // 通知报销人
        ExpenseBill bill = expenseBillMapper.selectById(approval.getBillId());
        if (bill != null) {
            sysMessageService.sendMessage(bill.getApplicantId(), "超预算审批驳回",
                "您的超预算报销单（" + approval.getBillNo() + "）被驳回，原因：" + reason);
        }
        return true;
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Boolean adjust(Long id, String adjustFromSubject, String adjustFromSubjectName,
                          BigDecimal adjustAmount, String opinion) {
        OverBudgetApproval approval = overBudgetApprovalMapper.selectById(id);
        validatePending(approval);

        if (adjustAmount.compareTo(approval.getShortageAmount()) < 0) {
            throw new ServiceException("调剂金额不得小于缺口金额");
        }

        // 校验调出科目余额
        BigDecimal fromBalance = budgetDetailMapper.getRemainingBudget(
            adjustFromSubject, approval.getCostCenterCode());
        if (fromBalance.compareTo(adjustAmount) < 0) {
            throw new ServiceException("调出科目可用余额不足");
        }

        // 调出科目扣减
        budgetDetailMapper.subtractBudgetBySubject(adjustFromSubject,
            approval.getCostCenterCode(), adjustAmount);
        // 调入科目增加
        budgetDetailMapper.addBudgetBySubject(approval.getBudgetSubject(),
            approval.getCostCenterCode(), adjustAmount);

        // 记录调剂流水
        BudgetAdjustmentLog adjLog = new BudgetAdjustmentLog();
        adjLog.setCostCenterCode(approval.getCostCenterCode());
        adjLog.setFromSubject(adjustFromSubject);
        adjLog.setFromSubjectName(adjustFromSubjectName);
        adjLog.setToSubject(approval.getBudgetSubject());
        adjLog.setToSubjectName(approval.getBudgetSubjectName());
        adjLog.setAdjustAmount(adjustAmount);
        adjLog.setRelatedApprovalId(id);
        adjLog.setRelatedBillNo(approval.getBillNo());
        adjLog.setOperatorId(LoginHelper.getUserId());
        adjLog.setOperatorName(LoginHelper.getNickName());
        adjLog.setOperateTime(new Date());
        adjustmentLogMapper.insert(adjLog);

        approval.setApprovalAction("ADJUST");
        approval.setAdjustFromSubject(adjustFromSubject);
        approval.setAdjustFromSubjectName(adjustFromSubjectName);
        approval.setAdjustAmount(adjustAmount);
        approval.setApprovalOpinion(opinion);
        approval.setApproverId(LoginHelper.getUserId());
        approval.setApproverName(LoginHelper.getNickName());
        approval.setApprovalTime(new Date());
        approval.setStatus("APPROVED");
        overBudgetApprovalMapper.updateById(approval);

        // 操作日志
        Map<String, Object> detail = new HashMap<>();
        detail.put("fromSubject", adjustFromSubject);
        detail.put("toSubject", approval.getBudgetSubject());
        detail.put("adjustAmount", adjustAmount);
        feeOperationLogService.log("OVER_BUDGET", "OB_ADJUST", approval.getBillId(),
            approval.getBillNo(), id, "PENDING", "APPROVED", JSONUtil.toJsonStr(detail));

        // 通知报销人
        ExpenseBill bill = expenseBillMapper.selectById(approval.getBillId());
        if (bill != null) {
            sysMessageService.sendMessage(bill.getApplicantId(), "预算调剂完成",
                "预算已调剂：调出科目" + adjustFromSubjectName + " → 调入科目" + approval.getBudgetSubjectName()
                    + "，金额 ¥" + adjustAmount);
        }
        return true;
    }

    private void validatePending(OverBudgetApproval approval) {
        if (approval == null) throw new ServiceException("审批单不存在");
        if (!"PENDING".equals(approval.getStatus())) {
            throw new ServiceException("该审批单已处理");
        }
    }
}
```

**apply() 逻辑**：
1. 校验报销单存在
2. 生成申请单号 BA-YYYYMMDD-XXX
3. 构建申请实体，currentLevel=1，status=DEPT_APPROVING
4. 关联部门预留预算ID
5. 写操作日志（BA_APPLY）
6. 通知部门领导

**approve() 逻辑（事务+悲观锁）**：
1. 校验申请单状态 + 当前用户角色匹配审批层级
2. 根据 currentLevel 确定使用哪个预留预算
3. SELECT FOR UPDATE 锁定预留预算行
4. 校验预留剩余额度 ≥ 申请金额
5. 更新预留预算 allocatedAmount
6. 更新目标科目可用余额（budget_preparation_detail）
7. 写入分摊记录（fee_reserved_allocation_log）
8. 更新申请单状态为 APPROVED
9. 写操作日志（BA_ALLOCATE）
10. 通知报销人
11. 预留预算预警检查

**escalate() 逻辑**：
1. currentLevel + 1，若 > 3 则抛异常
2. 更新 status：level 2 → BRANCH_APPROVING，level 3 → HQ_APPROVING
3. 关联对应层级预留预算ID
4. 写操作日志（BA_ESCALATE）
5. 通知上级审批人
